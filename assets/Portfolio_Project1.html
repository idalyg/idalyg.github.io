from sklearn.model_selection import GroupShuffleSplit

X = df.drop(columns=["Percent_Bleaching", "Sample_ID"])
y = df["Percent_Bleaching"]

splitter = GroupShuffleSplit(
    n_splits=1,
    test_size=0.2,
    random_state=42
)

train_idx, test_idx = next(
    splitter.split(X, y, groups=df["Sample_ID"])
)

X_train = X.iloc[train_idx]
X_test = X.iloc[test_idx]

y_train = y.iloc[train_idx]
y_test = y.iloc[test_idx]

from sklearn.cluster import KMeans
import sklearn.metrics

inertias = []
silouette_scores = []

for k in range(2, 9):
    kmeans = KMeans(n_clusters=k, random_state=42)
    kmeans.fit(X_train_processed)
    inertias.append(kmeans.inertia_)
    silouette_scores.append(sklearn.metrics.silhouette_score(X_train_processed, kmeans.labels_))

kmeans_1 = KMeans(n_clusters=1, random_state=42)
kmeans_1.fit(X_train_processed)
inertia_1 = kmeans_1.inertia_

plt.plot([1] + list(range(2, 9)), [inertia_1] + inertias, marker='o')
plt.xlabel("Number of Clusters (K)")
plt.ylabel("Inertia")
plt.title("Elbow Method for Coral Reef Clustering")
plt.show()
